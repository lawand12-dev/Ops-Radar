# COO Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade the COO chat from a single context-injected LLM call into a GPT-4o tool-use agent that dynamically fetches live data, streams visible tool-call steps to the UI, and queues actions for inline confirmation.

**Architecture:** The API route becomes a streaming SSE endpoint running an agent loop — GPT-4o decides which tools to call, the server executes them, results feed back into the model, repeat until `finish_reason === 'stop'`. The chat UI reads the SSE stream and renders tool steps and inline action cards as they arrive.

**Tech Stack:** GPT-4o function calling · Next.js App Router SSE (ReadableStream) · Supabase `agent_actions` table · TypeScript · existing integration adapters (Stripe, HubSpot, Harvest, ClickUp)

```typescript
// src/lib/coo/executor.ts
import { fetchStripeFinanceData } from '<npm>/lib/integrations/stripe/adapter';
import { fetchHubspotSalesData } from '<npm>/lib/integrations/hubspot/adapter';
import { fetchStripePerClientRevenue } from '<npm>/lib/integrations/stripe/profitability-adapter';
import { fetchHarvestPerClientHours } from '<npm>/lib/integrations/harvest/profitability-adapter';
import { computeClientProfitability } from '<npm>/lib/profitability';
import { fetchClickupProjectBudgets } from '<npm>/lib/integrations/clickup/scope-adapter';
import { getServerSupabaseClient } from '<npm>/lib/supabase-server';

export type ExecutorContext = {
  stripeKey: string | null;
  hubspotKey: string | null;
  harvestKey: string | null;
  clickupKey: string | null;
  blendedRate: number;
  teamSizeRaw: string;
  userId: string;
};

export type ToolResult =
  | { success: true; data: unknown }
  | { success: false; reason: string };

function teamSizeMin(raw: string): number | null {
  if (!raw) return null;
  if (raw === '1') return 1;
  const m = raw.match(/^(\d+)/);
  return m ? parseInt(m[1]!) : null;
}

async function runGetFinance(ctx: ExecutorContext): Promise<ToolResult> {
  if (!ctx.stripeKey) return { success: false, reason: 'Stripe is not connected. Go to Integrations to connect it.' };
  try {
    const data = await fetchStripeFinanceData(ctx.stripeKey);
    return {
      success: true,
      data: {
        invoiced_mtd_usd: data.invoiced,
        paid_mtd_usd: data.paid,
        overdue_usd: data.overdue,
        currency: data.currency,
      },
    };
  } catch {
    return { success: false, reason: 'Stripe fetch failed. The API key may have expired.' };
  }
}

async function runGetSales(ctx: ExecutorContext): Promise<ToolResult> {
  if (!ctx.hubspotKey) return { success: false, reason: 'HubSpot is not connected. Go to Integrations to connect it.' };
  try {
    const data = await fetchHubspotSalesData(ctx.hubspotKey);
    return {
      success: true,
      data: {
        pipeline_value_usd: data.pipeline_value,
        open_deals_count: data.open_deals_count,
        deals_won_this_month: data.deals_won_this_month,
        currency: data.currency,
      },
    };
  } catch {
    return { success: false, reason: 'HubSpot fetch failed. The API key may have expired.' };
  }
}

async function runGetProfitability(ctx: ExecutorContext): Promise<ToolResult> {
  if (!ctx.stripeKey) return { success: false, reason: 'Stripe is not connected — needed for revenue data.' };
  if (!ctx.harvestKey) return { success: false, reason: 'Harvest is not connected — needed for hours data.' };
  try {
    const [stripeData, harvestData] = await Promise.all([
      fetchStripePerClientRevenue(ctx.stripeKey),
      fetchHarvestPerClientHours(ctx.harvestKey),
    ]);
    const result = computeClientProfitability(stripeData, harvestData, ctx.blendedRate);
    return {
      success: true,
      data: {
        clients: result.clients.map((c) => ({
          name: c.name,
          revenue_mtd_usd: Math.round(c.revenue_mtd),
          hours_mtd: Math.round(c.hours_mtd * 10) / 10,
          cost_usd: Math.round(c.hours_mtd * ctx.blendedRate),
          margin_pct: Math.round(c.margin_pct),
        })),
      },
    };
  } catch {
    return { success: false, reason: 'Profitability fetch failed.' };
  }
}

async function runGetCapacity(ctx: ExecutorContext): Promise<ToolResult> {
  if (!ctx.harvestKey) return { success: false, reason: 'Harvest is not connected — needed for hours tracking.' };
  const minSize = teamSizeMin(ctx.teamSizeRaw);
  const capacityHours = minSize ? minSize * 160 : null;
  if (!capacityHours) return { success: false, reason: 'Team size is not set in account settings.' };
  try {
    const harvestData = await fetchHarvestPerClientHours(ctx.harvestKey);
    const totalHours = harvestData.clients.reduce((s, c) => s + c.hours_current, 0);
    const utilPct = Math.round((totalHours / capacityHours) * 100);
    return {
      success: true,
      data: {
        estimated_monthly_capacity_hours: capacityHours,
        hours_tracked_this_month: Math.round(totalHours * 10) / 10,
        utilization_pct: utilPct,
        available_hours: Math.round(capacityHours - totalHours),
      },
    };
  } catch {
    return { success: false, reason: 'Harvest capacity fetch failed.' };
  }
}

async function runGetScope(ctx: ExecutorContext): Promise<ToolResult> {
  if (!ctx.clickupKey) return { success: false, reason: 'ClickUp is not connected — needed for project budget data.' };
  if (!ctx.harvestKey) return { success: false, reason: 'Harvest is not connected — needed for hours logged data.' };
  try {
    const [clickupData, harvestData] = await Promise.all([
      fetchClickupProjectBudgets(ctx.clickupKey),
      fetchHarvestPerClientHours(ctx.harvestKey),
    ]);
    const harvestByName = Object.fromEntries(
      harvestData.clients.map((c) => [c.name.toLowerCase(), c.hours_current])
    );
    const projects = clickupData.projects.map((p) => {
      const logged = harvestByName[p.project_name.toLowerCase()] ?? 0;
      const overrunPct = p.estimated_hours > 0
        ? Math.round(((logged - p.estimated_hours) / p.estimated_hours) * 100)
        : 0;
      return {
        project_name: p.project_name,
        budget_hours: p.estimated_hours,
        logged_hours: Math.round(logged * 10) / 10,
        overrun_pct: overrunPct,
        in_scope: logged <= p.estimated_hours,
      };
    });
    return { success: true, data: { projects } };
  } catch {
    return { success: false, reason: 'Scope fetch failed.' };
  }
}

async function runQueueAction(
  args: Record<string, unknown>,
  ctx: ExecutorContext,
): Promise<ToolResult> {
  const supabase = getServerSupabaseClient();
  const { data, error } = await supabase
    .from('agent_actions')
    .insert({
      user_id: ctx.userId,
      action_type: args.action_type as string,
      payload: args.payload as Record<string, unknown>,
      status: 'pending',
    })
    .select('id')
    .single();

  if (error || !data) return { success: false, reason: 'Failed to save action.' };
  return { success: true, data: { action_id: data.id, label: args.label as string } };
}

export async function executeToolCall(
  name: string,
  args: Record<string, unknown>,
  ctx: ExecutorContext,
): Promise<ToolResult> {
  switch (name) {
    case 'get_finance':       return runGetFinance(ctx);
    case 'get_sales':         return runGetSales(ctx);
    case 'get_profitability': return runGetProfitability(ctx);
    case 'get_capacity':      return runGetCapacity(ctx);
    case 'get_scope':         return runGetScope(ctx);
    case 'queue_action':      return runQueueAction(args, ctx);
    default:
      return { success: false, reason: `Unknown tool: ${name}` };
  }
}
```

```typescript
// src/app/api/dashboard/coo/route.ts
import { type NextRequest, NextResponse } from 'next/server';
import { getCurrentUserId, getSessionSupabaseClient } from '<npm>/lib/supabase-ssr';
import { getServerSupabaseClient } from '<npm>/lib/supabase-server';
import { decryptApiKey } from '<npm>/lib/integrations/crypto';
import { applyRateLimit } from '<npm>/lib/rate-limit';
import { openai } from '<npm>/lib/openai';
import { canAccess, getUserPlan, upgradeRequiredBody } from '<npm>/lib/tier';
import { COO_TOOLS, TOOL_LABELS } from '<npm>/lib/coo/tools';
import { executeToolCall, type ExecutorContext } from '<npm>/lib/coo/executor';
import type OpenAI from 'openai';

export const dynamic = 'force-dynamic';
export const runtime = 'nodejs';
export const maxDuration = 60;

type UserMessage = { role: 'user'; content: string };
type AssistantMessage = { role: 'assistant'; content: string };
type IncomingMessage = UserMessage | AssistantMessage;

type AgentEvent =
  | { type: 'tool_call'; name: string; label: string }
  | { type: 'tool_done'; name: string }
  | { type: 'text_delta'; delta: string }
  | { type: 'action_proposed'; action_id: string; action_type: string; label: string }
  | { type: 'done' }
  | { type: 'error'; message: string };

const SUGGESTED_QUESTIONS = [
  'How did we do this month?',
  'Which clients should I worry about?',
  'Will I make payroll next month?',
  'Which clients should I reprice?',
  'How close are we to capacity?',
  'What are the most important things to do today?',
];

export async function GET(): Promise<NextResponse> {
  return NextResponse.json({ suggested_questions: SUGGESTED_QUESTIONS });
}

export async function POST(request: NextRequest): Promise<Response> {
  const userId = await getCurrentUserId();
  if (!userId) return NextResponse.json({ error: 'unauthorized' }, { status: 401 });

  const limited = await applyRateLimit(request, 'dashboard-coo', userId);
  if (limited) return limited as Response;

  const body = (await request.json().catch(() => ({}))) as { messages?: IncomingMessage[] };
  const messages = (body.messages ?? []).slice(-10);

  if (!messages.length || messages[messages.length - 1]?.role !== 'user') {
    return NextResponse.json({ error: 'invalid_messages' }, { status: 400 });
  }

  const service = getServerSupabaseClient();
  const { data: { user }, error: userErr } = await service.auth.admin.getUserById(userId);
  if (userErr || !user) return NextResponse.json({ error: 'not_found' }, { status: 404 });

  const meta = (user.user_metadata ?? {}) as Record<string, unknown>;
  const plan = getUserPlan(meta);
  if (!canAccess(plan, 'coo_chat')) {
    return NextResponse.json(upgradeRequiredBody('growth'), { status: 402 });
  }

  const blendedRate = typeof meta.blended_hourly_rate_usd === 'number' ? meta.blended_hourly_rate_usd : 85;
  const companyName = typeof meta.company_name === 'string' ? meta.company_name : 'your agency';
  const teamSizeRaw = typeof meta.team_size === 'string' ? meta.team_size : '';

  const session = await getSessionSupabaseClient();
  const { data: keyRows } = await session
    .from('integrations')
    .select('tool, encrypted_token, iv, auth_tag, status')
    .eq('client_id', userId)
    .eq('status', 'connected');

  const getKey = (tool: string): string | null => {
    const row = (keyRows ?? []).find((r) => r.tool === tool);
    try { return row ? decryptApiKey(row) : null; } catch { return null; }
  };

  const { data: pendingRows } = await session
    .from('agent_actions')
    .select('id')
    .eq('user_id', userId)
    .eq('status', 'pending');
  const pendingActionsCount = (pendingRows ?? []).length;

  const ctx: ExecutorContext = {
    stripeKey: getKey('stripe'),
    hubspotKey: getKey('hubspot'),
    harvestKey: getKey('harvest'),
    clickupKey: getKey('clickup'),
    blendedRate,
    teamSizeRaw,
    userId,
  };

  const systemPrompt = `You are the AI COO of ${companyName}. You answer questions about the business using live data from your tools.

Rules you MUST follow:
1. Before answering any question about metrics, finances, clients, pipeline, or capacity — call the relevant tool first
2. Base your answer ONLY on tool results — never guess or invent numbers
3. If a tool returns success: false, tell the user that data is unavailable and what integration to connect
4. When you spot a concrete problem (overdue invoice, unprofitable client, scope overrun) — call queue_action AFTER explaining it
5. Be concise and direct. Dollar amounts over adjectives. No markdown headers.

Account context (from settings — do not fetch via tools):
- Company: ${companyName}
- Blended hourly rate: $${blendedRate}/hr
- Team size: ${teamSizeRaw || 'not set in settings'}
- Pending actions awaiting approval: ${pendingActionsCount}`;

  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      function emit(event: AgentEvent) {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify(event)}\n\n`));
      }

      try {
        const history: OpenAI.Chat.Completions.ChatCompletionMessageParam[] = [
          ...messages.map((m) => ({ role: m.role, content: m.content } as OpenAI.Chat.Completions.ChatCompletionMessageParam)),
        ];

        let iterations = 0;
        const MAX_ITER = 5;

        while (iterations < MAX_ITER) {
          iterations++;

          const completion = await openai.chat.completions.create({
            model: 'gpt-4o',
            messages: [{ role: 'system', content: systemPrompt }, ...history],
            tools: COO_TOOLS,
            temperature: 0.2,
            max_tokens: 800,
          });

          const choice = completion.choices[0];
          if (!choice) break;

          if (choice.finish_reason === 'stop') {
            emit({ type: 'text_delta', delta: choice.message.content ?? '' });
            emit({ type: 'done' });
            controller.close();
            return;
          }

          if (choice.finish_reason === 'tool_calls') {
            const toolCalls = choice.message.tool_calls ?? [];
            history.push({
              role: 'assistant',
              content: choice.message.content ?? null,
              tool_calls: toolCalls,
            });

            for (const tc of toolCalls) {
              const name = tc.function.name;
              const args = JSON.parse(tc.function.arguments || '{}') as Record<string, unknown>;

              emit({ type: 'tool_call', name, label: TOOL_LABELS[name] ?? `Running ${name}...` });

              const result = await executeToolCall(name, args, ctx);

              emit({ type: 'tool_done', name });

              if (name === 'queue_action' && result.success) {
                const d = result.data as { action_id: string; label: string };
                emit({
                  type: 'action_proposed',
                  action_id: d.action_id,
                  action_type: args.action_type as string,
                  label: d.label,
                });
              }

              history.push({
                role: 'tool',
                tool_call_id: tc.id,
                content: JSON.stringify(result),
              });
            }
          }
        }

        emit({ type: 'error', message: 'The COO ran too many steps. Please try a more specific question.' });
        controller.close();
      } catch (err) {
        console.error('[coo-agent] error:', err);
        emit({ type: 'error', message: 'Something went wrong. Please try again.' });
        controller.close();
      }
    },
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache, no-transform',
      'X-Accel-Buffering': 'no',
    },
  });
}
```

```typescript
'use client';

import { useState, useRef, useEffect } from 'react';
import { Send, Loader2, CheckCircle, XCircle } from 'lucide-react';

type ToolStep = { name: string; label: string; done: boolean };

type ActionCard = {
  action_id: string;
  action_type: string;
  label: string;
  status: 'pending' | 'confirmed' | 'cancelled';
};

type UserMessage = { role: 'user'; content: string };
type AssistantMessage = {
  role: 'assistant';
  content: string;
  toolSteps: ToolStep[];
  actionCard: ActionCard | null;
};
type Message = UserMessage | AssistantMessage;

type AgentEvent =
  | { type: 'tool_call'; name: string; label: string }
  | { type: 'tool_done'; name: string }
  | { type: 'text_delta'; delta: string }
  | { type: 'action_proposed'; action_id: string; action_type: string; label: string }
  | { type: 'done' }
  | { type: 'error'; message: string };

const SUGGESTED: string[] = [
  'How did we do this month?',
  'Which clients should I worry about?',
  'Will I make payroll next month?',
  'Which clients should I reprice?',
  'How close are we to capacity?',
  'What should I do today?',
];

export function CooChat() {
  const [messages, setMessages] = useState<Message[]>([]);
  const [streaming, setStreaming] = useState<AssistantMessage | null>(null);
  const [input, setInput] = useState('');
  const [loading, setLoading] = useState(false);
  const bottomRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages, streaming, loading]);

  async function send(text: string) {
    const trimmed = text.trim();
    if (!trimmed || loading) return;

    const userMsg: UserMessage = { role: 'user', content: trimmed };
    const history = [...messages, userMsg];
    setMessages(history);
    setInput('');
    setLoading(true);

    const draft: AssistantMessage = { role: 'assistant', content: '', toolSteps: [], actionCard: null };
    setStreaming(draft);

    try {
      const res = await fetch('/api/dashboard/coo', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ messages: history.map((m) => ({ role: m.role, content: m.content })) }),
      });

      if (!res.ok || !res.body) {
        const data = await res.json().catch(() => ({})) as { error?: string };
        setStreaming(null);
        setMessages([...history, { role: 'assistant', content: data.error ?? 'Something went wrong.', toolSteps: [], actionCard: null }]);
        return;
      }

      const reader = res.body.getReader();
      const decoder = new TextDecoder();
      let buf = '';

      while (true) {
        const { done, value } = await reader.read();
        if (done) break;

        buf += decoder.decode(value, { stream: true });
        const lines = buf.split('\n');
        buf = lines.pop() ?? '';

        for (const line of lines) {
          if (!line.startsWith('data: ')) continue;
          const event = JSON.parse(line.slice(6)) as AgentEvent;

          if (event.type === 'tool_call') {
            setStreaming((prev) => {
              if (!prev) return prev;
              return { ...prev, toolSteps: [...prev.toolSteps, { name: event.name, label: event.label, done: false }] };
            });
          }

          if (event.type === 'tool_done') {
            setStreaming((prev) => {
              if (!prev) return prev;
              return {
                ...prev,
                toolSteps: prev.toolSteps.map((s) => s.name === event.name ? { ...s, done: true } : s),
              };
            });
          }

          if (event.type === 'text_delta') {
            setStreaming((prev) => {
              if (!prev) return prev;
              return { ...prev, content: prev.content + event.delta };
            });
          }

          if (event.type === 'action_proposed') {
            setStreaming((prev) => {
              if (!prev) return prev;
              return {
                ...prev,
                actionCard: {
                  action_id: event.action_id,
                  action_type: event.action_type,
                  label: event.label,
                  status: 'pending',
                },
              };
            });
          }

          if (event.type === 'done' || event.type === 'error') {
            setStreaming((prev) => {
              if (!prev) return prev;
              const finalContent = event.type === 'error' ? event.message : prev.content;
              const finalMsg: AssistantMessage = { ...prev, content: finalContent };
              setMessages((m) => [...m, finalMsg]);
              return null;
            });
          }
        }
      }
    } catch {
      setStreaming(null);
      setMessages([...history, { role: 'assistant', content: 'Connection error. Please try again.', toolSteps: [], actionCard: null }]);
    } finally {
      setLoading(false);
    }
  }

  async function confirmAction(actionId: string, msgIndex: number) {
    await fetch('/api/dashboard/actions', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action: 'approve', id: actionId }),
    });
    setMessages((prev) =>
      prev.map((m, i) => {
        if (i !== msgIndex || m.role !== 'assistant' || !m.actionCard) return m;
        return { ...m, actionCard: { ...m.actionCard, status: 'confirmed' } };
      })
    );
  }

  async function dismissAction(actionId: string, msgIndex: number) {
    await fetch('/api/dashboard/actions', {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action: 'dismiss', id: actionId }),
    });
    setMessages((prev) =>
      prev.map((m, i) => {
        if (i !== msgIndex || m.role !== 'assistant' || !m.actionCard) return m;
        return { ...m, actionCard: { ...m.actionCard, status: 'cancelled' } };
      })
    );
  }

  function handleKey(e: React.KeyboardEvent<HTMLTextAreaElement>) {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      void send(input);
    }
  }

  function renderAssistant(m: AssistantMessage, index: number) {
    return (
      <div key={index} className="flex justify-start">
        <div className="max-w-[85%] space-y-2">
          {m.toolSteps.map((step, si) => (
            <div
              key={si}
              className="flex items-center gap-2 rounded-[var(--radius)] border border-[var(--color-border)] bg-[var(--color-surface)] px-3 py-1.5 text-[12px] text-[var(--color-primary)]"
            >
              {step.done
                ? <CheckCircle size={12} className="shrink-0 text-emerald-400" />
                : <Loader2 size={12} className="shrink-0 animate-spin" />
              }
              {step.label}
            </div>
          ))}

          {m.content && (
            <div className="rounded-[var(--radius-lg)] border border-[var(--color-border)] bg-[var(--color-surface)] px-4 py-3 text-[13px] leading-[1.6] text-[var(--color-fg)]">
              {m.content}
            </div>
          )}

          {m.actionCard && (
            <div className="rounded-[var(--radius)] border border-emerald-500/30 bg-emerald-950/20 px-4 py-3">
              <p className="mb-2 text-[12px] text-emerald-300">{m.actionCard.label}</p>
              {m.actionCard.status === 'pending' && (
                <div className="flex gap-2">
                  <button
                    onClick={() => void confirmAction(m.actionCard!.action_id, index)}
                    className="rounded-[var(--radius)] bg-emerald-600 px-3 py-1 text-[12px] font-medium text-white hover:bg-emerald-500"
                  >
                    Confirm
                  </button>
                  <button
                    onClick={() => void dismissAction(m.actionCard!.action_id, index)}
                    className="rounded-[var(--radius)] border border-[var(--color-border)] px-3 py-1 text-[12px] text-[var(--color-fg-muted)] hover:text-[var(--color-fg)]"
                  >
                    Cancel
                  </button>
                </div>
              )}
              {m.actionCard.status === 'confirmed' && (
                <p className="flex items-center gap-1.5 text-[12px] text-emerald-400">
                  <CheckCircle size={12} /> Action confirmed
                </p>
              )}
              {m.actionCard.status === 'cancelled' && (
                <p className="flex items-center gap-1.5 text-[12px] text-[var(--color-fg-subtle)]">
                  <XCircle size={12} /> Cancelled
                </p>
              )}
            </div>
          )}
        </div>
      </div>
    );
  }

  const allMessages: Array<{ msg: Message; index: number }> = [
    ...messages.map((msg, index) => ({ msg, index })),
    ...(streaming ? [{ msg: streaming as Message, index: messages.length }] : []),
  ];

  return (
    <div className="flex h-full flex-col">
      <div className="flex-1 overflow-y-auto space-y-4 pb-4">
        {messages.length === 0 && !streaming && (
          <div className="pt-2">
            <p className="mb-4 text-[13px] text-[var(--color-fg-muted)]">
              Ask me anything about your agency — I have access to your live data.
            </p>
            <div className="grid grid-cols-1 gap-2 sm:grid-cols-2 lg:grid-cols-3">
              {SUGGESTED.map((q) => (
                <button
                  key={q}
                  onClick={() => void send(q)}
                  className="rounded-[var(--radius)] border border-[var(--color-border)] bg-[var(--color-surface)] px-3 py-2.5 text-left text-[12.5px] text-[var(--color-fg-muted)] hover:border-[var(--color-primary)] hover:text-[var(--color-fg)] transition-colors"
                >
                  {q}
                </button>
              ))}
            </div>
          </div>
        )}

        {allMessages.map(({ msg, index }) => {
          if (msg.role === 'user') {
            return (
              <div key={index} className="flex justify-end">
                <div className="max-w-[80%] rounded-[var(--radius-lg)] bg-[var(--color-primary)] px-4 py-3 text-[13px] leading-[1.6] text-white">
                  {msg.content}
                </div>
              </div>
            );
          }
          return renderAssistant(msg as AssistantMessage, index);
        })}

        <div ref={bottomRef} />
      </div>

      <div className="border-t border-[var(--color-border)] pt-3">
        <div className="flex items-end gap-2">
          <textarea
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={handleKey}
            placeholder="Ask your COO anything…"
            rows={1}
            disabled={loading}
            className="flex-1 resize-none rounded-[var(--radius)] border border-[var(--color-border)] bg-[var(--color-surface)] px-3 py-2.5 text-[13px] text-[var(--color-fg)] outline-none focus:border-[var(--color-primary)] placeholder:text-[var(--color-fg-subtle)] disabled:opacity-50"
            style={{ maxHeight: '120px', overflowY: 'auto' }}
          />
          <button
            onClick={() => void send(input)}
            disabled={loading || !input.trim()}
            className="flex h-9 w-9 shrink-0 items-center justify-center rounded-[var(--radius)] bg-[var(--color-primary)] text-white hover:opacity-90 disabled:opacity-40"
          >
            <Send size={14} />
          </button>
        </div>
        <p className="mt-1.5 text-[11px] text-[var(--color-fg-subtle)]">
          Enter to send · Shift+Enter for new line
        </p>
      </div>
    </div>
  );
}
```
